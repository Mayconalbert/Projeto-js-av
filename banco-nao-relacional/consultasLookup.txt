db.empresas.aggregate([
     {$match: {nome: "Vale"}},
     {$lookup: {
         from: "estado",
         localField: "cidadeId",
         foreignField: "cidades._id",
         as: "estados",
     }},
     {$unwind: "$estados"},
     {$unwind: "$estados.cidades"}, // separar cada cidade da lista em um documentos individual  
     {$addFields: {mesmaCidade: {$cmp: ["$estados.cidades._id", "$cidadeId"] } } }, // cria um campo 'mesmoCidade', que compara os dois valores ids e retonar '0 igual', '1 maior' e '-1 menor'
     {$match: { mesmaCidade: 0}}, filtra e traz apenas o documentos onde o resultado comparação for igual a 0 (ou seja, são iguais)
]).pretty()
{
        "_id" : ObjectId("6aa374c21edc66ebe001ce8d"),
        "nome" : "Vale",
        "cidadeId" : ObjectId("6a9b840b38f2f26d7cf58b3c"),
        "estados" : {
                "_id" : ObjectId("6a9b840b38f2f26d7cf58b3d"),
                "nome" : "São Paulo",
                "sigla" : "SP",
                "região" : "Sudeste",
                "cidades" : {
                        "_id" : ObjectId("6a9b840b38f2f26d7cf58b3c"),
                        "nome" : "Sorocaba",
                        "distancia" : 362,
                        "população" : 237748
                },
                "população" : 4789000
        },
        "mesmaCidade" : 0
}
